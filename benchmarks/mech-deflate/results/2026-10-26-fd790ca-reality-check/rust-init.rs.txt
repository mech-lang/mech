use std::{hint::black_box,time::Instant};
#[allow(dead_code)]
#[path="/private/tmp/mech-deflate-reality-check/src/core/tests/support/allocation_probe.rs"]
mod allocation_probe;
#[global_allocator]
static ALLOCATOR:allocation_probe::ProbeAllocator=allocation_probe::ProbeAllocator;
#[inline(never)]
fn initialize(n:usize,m:usize)->(Vec<u16>,Vec<u32>){(vec![0;n],vec![0;m])}
fn main(){
 let args:Vec<String>=std::env::args().collect();
 let n:usize=args[1].parse().unwrap();let m:usize=args[2].parse().unwrap();
 for rep in 0..8 {
  let start=Instant::now();
  let (result,allocations,bytes)=allocation_probe::measured_with_bytes(||black_box(initialize(black_box(n),black_box(m))));
  let ns=start.elapsed().as_nanos();
  assert_eq!(result.0.len(),262144);assert_eq!(result.1.len(),1500513);
  assert!(result.0.iter().all(|&x|x==0));assert!(result.1.iter().all(|&x|x==0));
  println!("{{\"engine\":\"rust\",\"case\":\"raw-buffer-initialization\",\"rep\":{rep},\"nanoseconds\":{ns},\"allocations\":{allocations},\"requested_bytes\":{bytes}}}");
 }
}
