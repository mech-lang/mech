use mech_engine::CanonicalSourceFrontend;
use mech_syntax::document::{AstNode,DocumentId,DocumentSyntax,ParseConfig,Revision,TextSnapshot,parse_canonical_document};
fn run(){
 let a:Vec<_>=std::env::args().collect();let source=std::fs::read_to_string(&a[1]).unwrap();
 let parsed=parse_canonical_document(TextSnapshot::new(DocumentId(0x50524f47),Revision(1),source.as_str()).unwrap(),ParseConfig::default());assert!(parsed.is_strictly_clean(),"{:?}",parsed.diagnostics);
 let syntax=DocumentSyntax::cast(parsed.syntax()).unwrap();let artifact=CanonicalSourceFrontend.compile_document_with_catalog(&syntax,mech_stdlib::source_catalog()).unwrap().compile_artifact().unwrap();
 let bytes=mech_engine::encode_program_artifact_bytecode_v1(&artifact).unwrap();let decoded=mech_engine::decode_program_artifact_bytecode_v1(&bytes).unwrap();assert_eq!(mech_engine::encode_program_artifact_bytecode_v1(&decoded).unwrap(),bytes);std::fs::write(&a[2],&bytes).unwrap();println!("{}",bytes.len());
}
fn main(){std::thread::Builder::new().stack_size(32*1024*1024).spawn(run).unwrap().join().unwrap();}
