+//! Diagnostic-only phase observation; never linked into latency candidates.
+use std::sync::{
+    Mutex, OnceLock,
+    atomic::{AtomicU64, Ordering::Relaxed},
+};
+use std::time::Instant;
+
+static ROOT: OnceLock<std::thread::ThreadId> = OnceLock::new();
+static POST: Mutex<Option<(Instant, Option<os::Snapshot>)>> = Mutex::new(None);
+static IN_PARALLEL: AtomicU64 = AtomicU64::new(0);
+static ROWS: Mutex<Vec<ParallelRow>> = Mutex::new(Vec::new());
+struct Counter {
+    calls: AtomicU64,
+    ns: AtomicU64,
+}
+impl Counter {
+    const fn new() -> Self {
+        Self {
+            calls: AtomicU64::new(0),
+            ns: AtomicU64::new(0),
+        }
+    }
+}
+static CANONICAL: [Counter; 3] = [Counter::new(), Counter::new(), Counter::new()];
+
+pub fn start() {
+    ROOT.set(std::thread::current().id()).unwrap();
+    let snapshot = os::Snapshot::read();
+    *POST.lock().unwrap() = Some((Instant::now(), snapshot));
+    begin_span(0, 8192, 63);
+}

+pub struct Canonical {
+    start: Instant,
+    bucket: usize,
+}
+impl Canonical {
+    pub fn new() -> Self {
+        let root = ROOT.get() == Some(&std::thread::current().id());
+        let bucket = if !root {
+            2
+        } else if IN_PARALLEL.load(Relaxed) != 0 {
+            1
+        } else {
+            0
+        };
+        Self {
+            start: Instant::now(),
+            bucket,
+        }
+    }
+}
+impl Drop for Canonical {
+    fn drop(&mut self) {
+        let counter = &CANONICAL[self.bucket];
+        counter
+            .ns
+            .fetch_add(self.start.elapsed().as_nanos().try_into().unwrap(), Relaxed);
+        counter.calls.fetch_add(1, Relaxed);
+    }
+}

+#[derive(serde::Serialize)]
+struct ParallelRow {
+    site: usize,
+    seconds: f64,
+    hardware: Option<os::Hardware>,
+}
+pub struct Parallel {
+    site: usize,
+    start: Instant,
+    before: Option<os::Snapshot>,
+}
+impl Parallel {
+    pub fn new(site: usize) -> Option<Self> {
+        if ROOT.get() != Some(&std::thread::current().id())
+            || IN_PARALLEL
+                .compare_exchange(0, 1, Relaxed, Relaxed)
+                .is_err()
+        {
+            return None;
+        }
+        let before = os::Snapshot::read();
+        Some(Self {
+            site,
+            start: Instant::now(),
+            before,
+        })
+    }
+}
+impl Drop for Parallel {
+    fn drop(&mut self) {
+        let seconds = self.start.elapsed().as_secs_f64();
+        let after = os::Snapshot::read();
+        let hardware = self.before.zip(after).and_then(|(a, b)| a.difference(b));
+        IN_PARALLEL.store(0, Relaxed);
+        ROWS.lock().unwrap().push(ParallelRow {
+            site: self.site,
+            seconds,
+            hardware,
+        });
+    }
+}

+    end_span();
+    let end = os::Snapshot::read();

+    #[repr(C)]
+    struct Timebase {
+        numer: u32,
+        denom: u32,
+    }
+    #[link(name = "proc")]
+    unsafe extern "C" {
+        fn proc_pid_rusage(pid: i32, flavor: i32, buffer: *mut std::ffi::c_void) -> i32;
+        fn getpid() -> i32;
+        fn mach_timebase_info(info: *mut Timebase) -> i32;
+    }
+    impl Snapshot {
+        pub fn read() -> Option<Self> {
+            let mut value = Self::default();
+            // SAFETY: caller-owned, correctly sized/aligned SDK structure.
+            let result = unsafe { proc_pid_rusage(getpid(), 4, (&mut value as *mut Self).cast()) };
+            (result == 0).then_some(value)
+        }
+        pub fn difference(self, end: Self) -> Option<Hardware> {
+            let mut timebase = Timebase { numer: 0, denom: 0 };
+            // ri_user_time and ri_system_time use Mach ticks, verified against
+            // getrusage timeval by the SDK C probe; convert rather than assume ns.
+            if unsafe { mach_timebase_info(&mut timebase) } != 0 || timebase.denom == 0 {
+                return None;
+            }
+            let seconds = f64::from(timebase.numer) / f64::from(timebase.denom) * 1e-9;
+            let instructions = end.instructions.checked_sub(self.instructions)?;
+            let cycles = end.cycles.checked_sub(self.cycles)?;
+            Some(Hardware {
+                instructions_retired: instructions,
+                cycles,
+                instructions_per_cycle: (cycles != 0)
+                    .then_some(instructions as f64 / cycles as f64),
+                user_cpu_seconds: end.user.checked_sub(self.user)? as f64 * seconds,
+                system_cpu_seconds: end.system.checked_sub(self.system)? as f64 * seconds,
+                resident_bytes_before: self.resident,
+                resident_bytes_after: end.resident,
+                physical_footprint_bytes_before: self.footprint,
+                physical_footprint_bytes_after: end.footprint,
+                lifetime_peak_physical_footprint_bytes_after: end.lifetime_peak_footprint,
+                pageins: end.pageins.checked_sub(self.pageins)?,
+            })
+        }
+    }
+    #[derive(Serialize)]

+        instructions_retired: u64,
+        cycles: u64,

+        user_cpu_seconds: f64,

+    }

+thread_local! {
+    static SPAN: std::cell::Cell<Option<(u8, usize, usize, Instant)>> = const { std::cell::Cell::new(None) };

+fn operation_bucket(operation: u64) -> usize {
+    if operation <= 2 {
+        operation as usize
+    } else if operation >= u64::MAX - 44 {
+        3 + (u64::MAX - operation) as usize
+    } else {
+        63
+    }
+}
+fn source_site(site: u64) -> usize {
+    if site < 8192 {
+        site as usize
+    } else {
+        SITE_OVERFLOWS.fetch_add(1, Relaxed);
+        8192
+    }
+}
+fn begin_span(kind: u8, site: usize, operation: usize) {
+    SPAN.set(Some((kind, site, operation, Instant::now())));
+}
+fn end_span() {
+    if let Some((kind, site, operation, start)) = SPAN.take() {
+        let ns = start.elapsed().as_nanos().try_into().unwrap();
+        let counter = match kind {
+            1 => &TAIL_NS,
+            2 => &SEGMENT_NS,
+            _ => &OTHER_NS,
+        };
+        counter[site].fetch_add(ns, Relaxed);
+        if kind == 0 {
+            OTHER_OP_NS[operation].fetch_add(ns, Relaxed);
+        }
+    }
+}
+pub(crate) fn helper_begin(site: u64, operation: u64) -> Option<Instant> {
+    let root = ROOT.get()?;
+    let key = operation_bucket(operation);
+    if root != &std::thread::current().id() {
+        WORKER_OPS[key].fetch_add(1, Relaxed);
+        return None;
+    }
+    end_span();
+    let _ = source_site(site);
+    ROOT_OPS[key].fetch_add(1, Relaxed);
+    Some(Instant::now())
+}
+pub(crate) fn helper_end(
+    start: Option<Instant>,
+    site: u64,
+    operation: u64,
+    next: Option<(u8, usize)>,
+) {
+    let Some(start) = start else {
+        return;
+    };
+    let key = operation_bucket(operation);
+    let site = source_site(site);
+    let ns = start.elapsed().as_nanos().try_into().unwrap();
+    HELPER_NS[key].fetch_add(ns, Relaxed);
+    HELPER_SITE_NS[site].fetch_add(ns, Relaxed);
+    let (kind, next_site) = next.unwrap_or((0, site));
+    begin_span(kind, source_site(next_site as u64), key);
+}
